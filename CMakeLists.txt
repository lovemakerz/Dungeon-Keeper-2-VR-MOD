cmake_minimum_required(VERSION 3.21)
project(DK2VR VERSION 0.9.24 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
if(MSVC)
  add_compile_options(/W4 /EHsc /utf-8)
endif()
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
  add_executable(DK2VRHost src/host/Host.cpp)
  target_include_directories(DK2VRHost PRIVATE research/OpenXR-SDK/include)
  target_link_libraries(DK2VRHost PRIVATE
    "${CMAKE_CURRENT_SOURCE_DIR}/build/openxr/src/loader/Release/openxr_loader.lib"
    d3d11 dxgi d3dcompiler advapi32 shlwapi shell32 ole32)
else()
  add_library(DK2VR SHARED src/plugin/Plugin.cpp src/plugin/exports.def)
  target_link_libraries(DK2VR PRIVATE d3d9)
  add_executable(DK2VRInspect tools/InspectProcess.cpp)
  target_link_libraries(DK2VRInspect PRIVATE dbghelp)
  add_executable(DK2VRDebug tools/DebugLaunch.cpp)
  target_link_libraries(DK2VRDebug PRIVATE dbghelp)
endif()
enable_testing()
add_executable(DK2VRMathTests tests/MathTests.cpp)
add_test(NAME StereoGeometryAndProtocol COMMAND DK2VRMathTests)
add_executable(DK2VRTableTests tests/TableTests.cpp)
add_test(NAME RectangularTableGeometry COMMAND DK2VRTableTests)
add_executable(DK2VRTableGpuTests tests/TableGpuTests.cpp)
target_link_libraries(DK2VRTableGpuTests PRIVATE d3d9)
add_executable(DK2VRGpuTest tests/GpuTransfer.cpp)
target_link_libraries(DK2VRGpuTest PRIVATE d3d9 d3d11 dxgi)
add_executable(DK2VRAnchorTests tests/AnchorTests.cpp)
add_test(NAME DemeoTableAnchor COMMAND DK2VRAnchorTests)
add_executable(DK2VRGrabTests tests/GrabTests.cpp)
add_test(NAME TableGrabGestures COMMAND DK2VRGrabTests)
add_executable(DK2VRDecoupleTests tests/DecoupleTests.cpp)
add_test(NAME DecoupledRenderingTransforms COMMAND DK2VRDecoupleTests)
